function doGet(e) { var lock = LockService.getScriptLock(); if (!lock.tryLock(10000)) { return ContentService.createTextOutput("系統忙碌中，請稍後再試。"); } try { var action = e.parameter.action || "borrow"; var ss = SpreadsheetApp.getActiveSpreadsheet(); // ========================================== // 動作 0：使用者登入驗證 (供 login.html 使用) // ========================================== if (action === "login") { var user = e.parameter.user ? String(e.parameter.user).trim() : ""; var pass = e.parameter.pass ? String(e.parameter.pass).trim() : ""; if (!user || !pass) { return ContentService.createTextOutput(JSON.stringify({ success: false, msg: "請輸入帳號與密碼" })) .setMimeType(ContentService.MimeType.JSON); } var userSheet = ss.getSheetByName("使用者"); if (!userSheet) { return ContentService.createTextOutput(JSON.stringify({ success: false, msg: "找不到「使用者」工作表" })) .setMimeType(ContentService.MimeType.JSON); } var users = userSheet.getDataRange().getValues(); for (var u = 1; u < users.length; u++) { if (String(users[u][0]).trim() === user && String(users[u][1]).trim() === pass) { return ContentService.createTextOutput(JSON.stringify({ success: true, name: String(users[u][2]).trim() || user })).setMimeType(ContentService.MimeType.JSON); } } return ContentService.createTextOutput(JSON.stringify({ success: false, msg: "帳號或密碼錯誤！" })) .setMimeType(ContentService.MimeType.JSON); } // 取得設備清單工作表 var sheet = ss.getSheetByName("工作表1"); var data = sheet.getDataRange().getValues(); // ========================================== // 動作 1：取得所有設備清單 (供 list.html 使用) // ========================================== if (action === "list") { var list = []; for (var j = 1; j < data.length; j++) { if (data[j][0]) { list.push({ id: String(data[j][0]).trim(), name: String(data[j][1]).trim(), status: String(data[j][2]).trim(), note: String(data[j][3]).trim(), brand: String(data[j][4]).trim() }); } } return ContentService.createTextOutput(JSON.stringify(list)) .setMimeType(ContentService.MimeType.JSON); } // ========================================== // 動作 2：新增設備 (供 add.html 使用) // ========================================== if (action === "add") { var newId = e.parameter.id ? String(e.parameter.id).trim() : ""; var newName = e.parameter.name ? String(e.parameter.name).trim() : ""; var newBrand = e.parameter.brand ? String(e.parameter.brand).trim() : ""; if (!newId || !newName) { return ContentService.createTextOutput("【新增失敗】請輸入設備編號與名稱！"); } for (var k = 1; k < data.length; k++) { if (String(data[k][0]).trim() === newId) { return ContentService.createTextOutput("【新增失敗】設備編號「" + newId + "」已存在！"); } } sheet.appendRow([newId, newName, "可借用", "", newBrand]); return ContentService.createTextOutput("【新增成功】設備「" + newName + "」已成功加入清單！"); } // ========================================== // 動作 3：借用 / 歸還 操作 (供 index.html 使用) // ========================================== var equipmentId = e.parameter.id; var userName = e.parameter.userName ? String(e.parameter.userName).trim() : "未記錄使用者"; if (!equipmentId) { return ContentService.createTextOutput("請提供設備 ID，例如：?id=1001"); } for (var i = 1; i < data.length; i++) { if (String(data[i][0]).trim() === String(equipmentId).trim()) { var currentStatus = String(data[i][2]).trim(); var eqName = data[i][1]; var eqBrand = data[i][4]; // --- 借用邏輯 --- if (action === "borrow") { if (currentStatus === "可借用") { var now = Utilities.formatDate(new Date(), "Asia/Taipei", "yyyy/MM/dd HH:mm"); var noteText = userName + " (" + now + ")"; sheet.getRange(i + 1, 3).setValue("借出中"); sheet.getRange(i + 1, 4).setValue(noteText); return ContentService.createTextOutput("【借用成功】\n" + eqBrand + " " + eqName + "\n借用人：" + userName); } else { return ContentService.createTextOutput("【借用失敗】該設備目前狀態為：" + currentStatus); } } // --- 歸還邏輯 --- else if (action === "return") { if (currentStatus === "借出中") { sheet.getRange(i + 1, 3).setValue("可借用"); sheet.getRange(i + 1, 4).setValue(""); return ContentService.createTextOutput("【歸還成功】\n" + eqBrand + " " + eqName + " 已歸還。"); } else { return ContentService.createTextOutput("【歸還失敗】該設備未借出，目前狀態為：" + currentStatus); } } else { return ContentService.createTextOutput("未知的操作指令！"); } } } return ContentService.createTextOutput("找不到設備編號：" + equipmentId); } finally { lock.releaseLock(); } }
